{# Macros that turn the parts of a song of songs.json into HTML. The shape of the data is tools/songs.schema.json of the songbook repository; the chord rule is D-123, set in static/songbook.css. The HTML of a line has no whitespace between its items on purpose: the spaces of the song are in the text itself, and any other would be shown as one. #} {# A text: a list of runs, gaps (pauses, in em) and footnote marks. #} {% macro text(parts) -%} {%- for part in parts -%} {%- if part.type == "run" -%} {%- if part.style -%} {{ part.text }} {%- else -%}{{ part.text }}{%- endif -%} {%- elif part.type == "gap" -%} {%- else -%} {{ part.number }} {%- endif -%} {%- endfor -%} {%- endmacro %} {# A chord: a list of parts, the quality and the bass note being raised or lowered. #} {% macro chord(parts) -%} {%- for part in parts -%} {%- if part.position == "superscript" -%}{{ part.text }} {%- elif part.position == "subscript" -%}{{ part.text }} {%- else -%}{{ part.text }}{%- endif -%} {%- endfor -%} {%- endmacro %} {# A chord and the text under it. `endsInWord`: the next chord is inside the word, so the gap before it holds a hyphen when it is wide enough. #} {% macro segment(item) -%} {%- if item.chord -%}{{ chord(item.chord) }}{%- endif -%} {%- if item.endsInWord -%} {{ text(item.lyrics) }} {%- else -%}{{ text(item.lyrics) }}{%- endif -%} {%- endmacro %} {# The things on a line between and beside the segments. #} {% macro line_item(item) -%} {%- if item.type == "segment" -%}{{ segment(item) }} {%- elif item.type == "repeatStart" -%}\: {%- elif item.type == "repeatEnd" -%}:\ {%- elif item.type == "cue" -%}{{ item.label }} {%- elif item.type == "note" -%}{{ text(item.content) }} {%- elif item.type == "speaker" -%}{{ item.name }}: {%- elif item.type == "score" -%} {%- endif -%} {%- endmacro %} {# A line of chords only is set in the text row, as in the book. #} {% macro line(item) -%}
{%- for part in item["items"] -%}{{ line_item(part) }}{%- endfor -%}
{%- endmacro %} {# One block of a song: the label in its own column, the lines beside it. #} {% macro block(label, lines, classes) -%}
{{ label }}
{%- for item in lines %} {{ line(item) }} {%- endfor %}
{%- endmacro %} {% macro body_item(item) -%} {%- if item.type == "block" -%}{{ block(item.label, item.lines, "block-" ~ item.kind) }} {%- elif item.type == "speakerBlock" -%}{{ block(item.name ~ ":", item.lines, "block-speaker") }} {%- elif item.type == "cueBlock" -%}{{ block(item.label, [item.line], "block-cue") }} {%- elif item.type == "note" -%}

{{ text(item.content) }}

{%- elif item.type == "rule" -%}
{%- endif -%} {%- endmacro %}
