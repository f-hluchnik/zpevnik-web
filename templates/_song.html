{# Macros that turn the parts of a song of songs.json into HTML. The shape of the data is tools/songs.schema.json of the songbook repository; the chord rule is D-123, set in static/songbook.css. The HTML of a line has no whitespace between its items on purpose: the spaces of the song are in the text itself, and any other would be shown as one. #} {# A text: a list of runs, gaps (pauses, in em) and footnote marks. #} {% macro text(parts) -%} {%- for part in parts -%} {%- if part.type == "run" -%} {%- if part.style -%} {{ part.text }} {%- else -%}{{ part.text }}{%- endif -%} {%- elif part.type == "gap" -%} {%- else -%} {{ part.number }} {%- endif -%} {%- endfor -%} {%- endmacro %} {# A chord: a list of parts, the quality and the bass note being raised or lowered. #} {% macro chord(parts) -%} {%- for part in parts -%} {%- if part.position == "superscript" -%}{{ part.text }} {%- elif part.position == "subscript" -%}{{ part.text }} {%- else -%}{{ part.text }}{%- endif -%} {%- endfor -%} {%- endmacro %} {# A chord and the text under it (D-123, D-176). The chord's column holds the head of the text, the shortest run of whole words that covers the chord; the tail flows after it and wraps at any space, as the book's lines do. Text with no chord flows too. `endsInWord`: the next chord is inside the word. Text that flows gets a soft hyphen there, so a line that breaks there shows a hyphen as the book does; a column goes to the next row together with the next column (word_groups), and its gap holds a hyphen when it is wide enough. #} {% macro column(item) -%} {%- set head, tail = item | split_head -%} {{ chord(item.chord) }} {%- if item.endsInWord and not tail -%} {{ text(head) }} {%- else -%}{{ text(head) }}{%- endif -%} {%- endmacro %} {% macro flow(parts, ends_in_word) -%} {{ text(parts) }}{% if ends_in_word %}­{% endif %} {%- endmacro %} {% macro tail(item) -%} {%- set head, tail = item | split_head -%} {%- if tail -%}{{ flow(tail, item.endsInWord) }}{%- endif -%} {%- endmacro %} {% macro segment(item) -%} {%- if item.chord -%}{{ column(item) }}{{ tail(item) }} {%- else -%}{{ flow(item.lyrics, item.endsInWord) }}{%- endif -%} {%- endmacro %} {# The things on a line between and beside the segments. #} {% macro line_item(item) -%} {%- if item.type == "segment" -%}{{ segment(item) }} {%- elif item.type == "repeatStart" -%}\: {%- elif item.type == "repeatEnd" -%}:\ {%- elif item.type == "cue" -%}{{ item.label }} {%- elif item.type == "note" -%}{{ text(item.content) }} {%- elif item.type == "speaker" -%}{{ item.name }}: {%- elif item.type == "score" -%} {%- endif -%} {%- endmacro %} {# A line of chords only is set in the text row, as in the book. A word split by chords between columns is one group (word_groups) that moves to the next row whole. #} {% macro line(item) -%}
{%- for group in item["items"] | word_groups -%} {%- if group | length > 1 -%} {% for part in group %}{{ column(part) }}{% endfor %}{{ tail(group[-1]) }} {%- else -%}{{ line_item(group[0]) }}{%- endif -%} {%- endfor -%}
